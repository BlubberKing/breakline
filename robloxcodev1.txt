local function approach(a, b, speed, dt) return a + (b - a) * (1 - exp(-speed * dt)) end
local function poseCF(p, k)
	return CFrame.new(p[1] * k) * CFrame.fromEulerAnglesYXZ(rad(p[2].X), rad(p[2].Y), rad(p[2].Z))
end
local function playAt(part, key, volume)
	local id = SoundBank and SoundBank[key]
	if type(id) ~= "string" or id == "" then return end
	local s = Instance.new("Sound")
	s.SoundId = id
	s.Volume = volume or 0.6
	s.RollOffMaxDistance = 120
	s.Parent = part
	s:Play()
	Debris:AddItem(s, 4)
end

local states, retryAt = {}, {}

local function setup(char)
	local hum = char:FindFirstChildOfClass("Humanoid")
	local root = char:FindFirstChild("HumanoidRootPart")
	local ut, lt, head = char:FindFirstChild("UpperTorso"), char:FindFirstChild("LowerTorso"), char:FindFirstChild("Head")
	local gun = char:FindFirstChild("TPGun")
	if not (hum and root and ut and lt and head and gun and gun.PrimaryPart) then return nil end
	local waist, neck, rootJ = ut:FindFirstChild("Waist"), head:FindFirstChild("Neck"), lt:FindFirstChild("Root")
	local joint = ut:FindFirstChild("GunJoint")
	local rua, rh = char:FindFirstChild("RightUpperArm"), char:FindFirstChild("RightHand")
	local lua, lh = char:FindFirstChild("LeftUpperArm"), char:FindFirstChild("LeftHand")
	local rGrip, lFore, lMag = gun:FindFirstChild("RightGrip", true), gun:FindFirstChild("LeftForegrip", true), gun:FindFirstChild("LeftMagwell", true)
	if not (waist and neck and rootJ and joint and rua and rh and lua and lh and rGrip and lFore and lMag) then return nil end

	local k = ut.Size.Y / 1.6
	local created = {}
	local function att(parent, name, pos)
		local a = Instance.new("Attachment")
		a.Name = name
		a.Position = pos * k
		a.Parent = parent
		table.insert(created, a)
		return a
	end
	local function ik(name, chainRoot, eff, target, pole)
		local c = Instance.new("IKControl")
		c.Name = name
		c.Type = Enum.IKControlType.Transform
		c.ChainRoot, c.EndEffector, c.Target, c.Pole = chainRoot, eff, target, pole
		c.Weight = 1
		c.SmoothTime = 0
		c.Parent = hum
		table.insert(created, c)
		return c
	end

	local st = {
		char = char, hum = hum, root = root, k = k, created = created,
		waist = waist, neck = neck, rootJ = rootJ, joint = joint,
		waistC0 = waist.C0, neckC0 = neck.C0, rootC0 = rootJ.C0,
		rHip = (char:FindFirstChild("RightUpperLeg") or {}).RightHip,
		lHip = (char:FindFirstChild("LeftUpperLeg") or {}).LeftHip,
		rKnee = (char:FindFirstChild("RightLowerLeg") or {}).RightKnee,
		lKnee = (char:FindFirstChild("LeftLowerLeg") or {}).LeftKnee,
		lFore = lFore, lMag = lMag, gun = gun,
		shoulder = CFG.SHOULDER * k, poses = {},
		pitch = 0, ads = 0, sprint = 0, reload = 0, reloadT = 0, slide = 0, mantle = 0,
		kick = 0, bobT = 0, turnLag = 0, flashToken = 0, wasSlide = false, restored = true,
	}
	for name, p in pairs(CFG.POSES) do st.poses[name] = poseCF(p, k) end

	st.pouch = att(lt, "TP_Pouch", CFG.POUCH)
	st.ikR = ik("TP_IK_Right", rua, rh, rGrip, att(ut, "TP_RightPole", CFG.RIGHT_POLE))
	st.ikL = ik("TP_IK_Left", lua, lh, lFore, att(ut, "TP_LeftPole", CFG.LEFT_POLE))

	st.magParts = {}
	for _, d in ipairs(gun:GetDescendants()) do
		if d:IsA("BasePart") and d:GetAttribute("IsMag") then table.insert(st.magParts, d) end
	end
	local mag = gun:FindFirstChild("Mag")
	if mag then
		local hm = mag:Clone()
		hm:ClearAllChildren()
		hm.Name = "TP_HandMag"
		hm.Transparency = 1
		local w = Instance.new("Weld")
		w.Part0, w.Part1, w.C0 = lh, hm, CFrame.new(0.12, -0.1, 0)
		w.Parent = hm
		hm.Parent = char
		st.handMag = hm
		table.insert(created, hm)
	end

	char.AncestryChanged:Connect(function(_, parent)
		if not parent then states[char] = nil end
	end)
	states[char] = st
	return st
end

local function getState(char)
	local st = states[char]
	if st then
		if not st.gun.Parent then states[char] = nil else return st end
	end
	if (retryAt[char] or 0) > os.clock() then return nil end
	st = setup(char)
	if not st then retryAt[char] = os.clock() + 0.5 end
	return st
end

local function restore(st)
	if st.restored then return end
	st.restored = true
	st.waist.C0, st.neck.C0, st.rootJ.C0 = st.waistC0, st.neckC0, st.rootC0
	st.ikR.Weight, st.ikL.Weight = 0, 0
end

------------------------------------------------------------------ per-frame pose
local function update(st, dt, t, isLocal)
	if st.hum.Health <= 0 then restore(st) return end
	st.restored = false
	st.ikR.Weight = 1

	st.pitch  = approach(st.pitch, t.pitch, 18, dt)
	st.ads    = approach(st.ads, (t.ads and not t.mantle) and 1 or 0, 15, dt)
	st.sprint = approach(st.sprint, (t.sprint and not t.ads and not t.slide) and 1 or 0, 9, dt)
	st.slide  = approach(st.slide, t.slide and 1 or 0, 14, dt)
	st.mantle = approach(st.mantle, t.mantle and 1 or 0, 18, dt)
	st.reload = approach(st.reload, t.reloading and 1 or 0, 12, dt)
	st.reloadT = t.reloading and (st.reloadT + dt) or 0
	st.kick = st.kick * exp(-dt * 16)

	if t.slide and not st.wasSlide and not isLocal then playAt(st.root, "Slide", 0.7) end
	st.wasSlide = t.slide

	local k = st.k
	-- gun lags slightly behind body turns
	local lv = st.root.CFrame.LookVector
	local yaw = math.atan2(-lv.X, -lv.Z)
	local d = yaw - (st.lastYaw or yaw)
	if d > math.pi then d -= 2 * math.pi elseif d < -math.pi then d += 2 * math.pi end
	st.lastYaw = yaw
	local lagTarget = clamp(-(d / math.max(dt, 1 / 240)) * CFG.TURN_LAG_DEG, -CFG.TURN_LAG_MAX, CFG.TURN_LAG_MAX)
	st.turnLag = approach(st.turnLag, lagTarget, 10, dt)

	-- walk bob
	local vel = st.root.AssemblyLinearVelocity
	local hs = V3(vel.X, 0, vel.Z).Magnitude
	local onGround = st.hum.FloorMaterial ~= Enum.Material.Air
	st.bobT += dt * hs * 0.55
	local amp = clamp(hs / 16, 0, 1.5) * (1 - 0.75 * st.ads) * (1 - st.slide) * (onGround and 1 or 0.25)
	local bob = CFrame.new(sin(st.bobT) * CFG.BOB * amp, -abs(cos(st.bobT)) * CFG.BOB * amp, 0)
		* CFrame.Angles(0, 0, sin(st.bobT) * rad(2.5) * amp)

	-- blended pose
	local P = st.poses
	local pose = P.Hip:Lerp(P.ADS, st.ads):Lerp(P.Slide, st.slide):Lerp(P.Sprint, st.sprint)
		:Lerp(P.Reload, st.reload):Lerp(P.Mantle, st.mantle)
	local lowered = math.max(st.sprint, st.mantle)
	local gunPitch = st.pitch * (1 - CFG.TORSO_SHARE) * (1 - 0.75 * lowered)
	local S = st.shoulder
	local aimCF = CFrame.new(S) * CFrame.Angles(gunPitch, rad(st.turnLag), 0) * CFrame.new(-S)
	local kick = CFrame.new(0, 0, CFG.KICK_BACK * st.kick * k) * CFrame.Angles(rad(CFG.KICK_UP_DEG * st.kick), 0, 0)
	st.joint.C0 = aimCF * pose * bob * kick

	-- body
	local torsoPitch = st.pitch * CFG.TORSO_SHARE * (1 - 0.5 * lowered) + rad(CFG.KICK_TORSO_DEG * st.kick) - rad(12) * st.slide
	st.waist.C0 = st.waistC0 * CFrame.Angles(torsoPitch, 0, 0)
	st.neck.C0 = st.neckC0 * CFrame.Angles(st.pitch * CFG.NECK_SHARE - rad(14) * st.slide, 0, 0)
	st.rootJ.C0 = st.rootC0 * CFrame.Angles(rad(CFG.SLIDE_LEAN) * st.slide, 0, 0)

	-- left hand: foregrip, or reload sequence (mag -> pouch -> mag -> foregrip)
	local target, smooth = st.lFore, 0
	local phase = st.reloadT / CFG.RELOAD_TIME
	if t.reloading then
		smooth = 0.09
		if phase < 0.1 then target = st.lFore
		elseif phase < 0.38 then target = st.lMag
		elseif phase < 0.62 then target = st.pouch
		elseif phase < 0.86 then target = st.lMag
		end
	end
	if st.ikL.Target ~= target then st.ikL.Target = target end
	st.ikL.SmoothTime = smooth
	st.ikL.Weight = 1 - 0.9 * st.mantle

	local magOffGun = t.reloading and phase >= 0.36 and phase < 0.7
	for _, p in ipairs(st.magParts) do p.Transparency = magOffGun and 1 or 0 end
	if st.handMag then st.handMag.Transparency = (t.reloading and phase >= 0.5 and phase < 0.7) and 0 or 1 end
end

local localT = { pitch = 0, ads = false, sprint = false, slide = false, mantle = false, reloading = false }
local lastSend, sent = 0, {}

local function isFirstPerson(char)
	local head = char and char:FindFirstChild("Head")
	return head and (workspace.CurrentCamera.CFrame.Position - head.Position).Magnitude < 1.5
end

RunService.Heartbeat:Connect(function()
	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return end
	local cam = workspace.CurrentCamera
	localT.pitch = math.asin(clamp(cam.CFrame.LookVector.Y, -1, 1))
	localT.slide = char:GetAttribute("Sliding") == true
	localT.mantle = char:GetAttribute("Mantling") == true
	localT.ads = cam.FieldOfView < CFG.ADS_FOV_BELOW and not localT.mantle
	localT.sprint = not localT.ads and hum.MoveDirection.Magnitude > 0.1 and hum.WalkSpeed >= CFG.SPRINT_SPEED_ABOVE
	localT.reloading = char:GetAttribute("Reloading") == true

	local now = os.clock()
	if now - lastSend < 1 / CFG.SEND_RATE then return end
	local changed = abs(localT.pitch - (sent.pitch or 0)) > 0.012 or localT.ads ~= sent.ads or localT.sprint ~= sent.sprint
		or localT.slide ~= sent.slide or localT.mantle ~= sent.mantle
	if changed or now - lastSend > 0.5 then
		Look:FireServer(localT.pitch, localT.ads, localT.sprint, localT.slide, localT.mantle)
		sent = { pitch = localT.pitch, ads = localT.ads, sprint = localT.sprint, slide = localT.slide, mantle = localT.mantle }
		lastSend = now
	end
end)

local function remoteTarget(char)
	return {
		pitch = char:GetAttribute("AimPitch") or 0,
		ads = char:GetAttribute("ADS") == true,
		sprint = char:GetAttribute("Sprint") == true,
		slide = char:GetAttribute("Sliding") == true,
		mantle = char:GetAttribute("Mantling") == true,
		reloading = char:GetAttribute("Reloading") == true,
	}
end

local poseSignal = (ok and preAnim) or RunService.Stepped

-- Runs before animation + IK, so the IK solves against the new gun/torso pose
poseSignal:Connect(function(a, b)
	local dt = (typeof(b) == "number") and b or a
	for _, plr in ipairs(Players:GetPlayers()) do
		local char = plr.Character
		if char then
			local st = getState(char)
			if st then
				if plr == player then
					if isFirstPerson(char) then restore(st) else update(st, dt, localT, true) end
				else
					update(st, dt, remoteTarget(char), false)
				end
			end
		end
	end
end)

-- Leg pose for sliding (after the Animator so it overrides walk/run)
RunService.Stepped:Connect(function()
	for _, st in pairs(states) do
		if st.slide > 0.01 and not st.restored and st.rHip and st.lHip and st.rKnee and st.lKnee then
			local a = st.slide
			st.rHip.Transform = st.rHip.Transform:Lerp(CFrame.Angles(rad(60), 0, 0), a)
			st.rKnee.Transform = st.rKnee.Transform:Lerp(CFrame.Angles(rad(-12), 0, 0), a)
			st.lHip.Transform = st.lHip.Transform:Lerp(CFrame.Angles(rad(18), 0, rad(-8)), a)
			st.lKnee.Transform = st.lKnee.Transform:Lerp(CFrame.Angles(rad(-100), 0, 0), a)
		end
	end
end)

Shot.OnClientEvent:Connect(function(char)
	if typeof(char) ~= "Instance" or char == player.Character or not char.Parent then return end
	local st = getState(char)
	if st then st.kick = math.min(st.kick + 1, 2.5) end

	local gun = char:FindFirstChild("TPGun")
	if not gun then return end
	local flash = gun:FindFirstChild("Flash")
	local muzzle = gun:FindFirstChild("Muzzle")
	local light = muzzle and muzzle:FindFirstChild("FlashLight")
	local token = (st and st.flashToken or 0) + 1
	if st then st.flashToken = token end
	local sc = gun:GetAttribute("Scale") or 1
	if flash then
		flash.Transparency = 0.1
		local s = (0.3 + math.random() * 0.15) * sc * 1.6
		flash.Size = V3(s, s, (0.5 + math.random() * 0.35) * sc * 1.6)
	end
	if light then light.Enabled = true end
	task.delay(0.045, function()
		if st and st.flashToken ~= token then return end
		if flash then flash.Transparency = 1 end
		if light then light.Enabled = false end
	end)
end)

-- ================= HELPERS =================
local function approach(a, target, dt, time)
	return a + (target - a) * (1 - math.exp(-dt * 4 / time))
end
local function seg(u, a, b) return math.clamp((u - a) / (b - a), 0, 1) end
local function smooth(x) return x * x * (3 - 2 * x) end
local function rnd() return math.random() * 2 - 1 end

local function updateHUD()
	ammoL.Text = tostring(ammo)
	ammoL.TextColor3 = (ammo <= cfg.MagSize * 0.25) and Color3.fromRGB(255, 80, 80) or Color3.new(1, 1, 1)
	nameL.Text = reloading and "RELOADING" or cfg.Name
	if not reloading then
		barFill.BackgroundColor3 = Color3.new(1, 1, 1)
		barFill.Size = UDim2.fromScale(ammo / cfg.MagSize, 1)
	end
end
updateHUD()

local function enemyHumanoid(model)
	local hum = model and model:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 or model == character then return nil end
	local p = Players:GetPlayerFromCharacter(model)
	if p and not player.Neutral and p.Team == player.Team then return nil end
	return hum
end

local function currentSpread()
	local s = cfg.SpreadHip + (cfg.SpreadADS - cfg.SpreadHip) * aimAlpha
	if humanoid.FloorMaterial == Enum.Material.Air then s += cfg.SpreadAir
	elseif humanoid.MoveDirection.Magnitude > 0.1 then s += cfg.SpreadMove * (1 - aimAlpha * 0.5) end
	return s + bloom
end

local function spreadDirection(dir, deg)
	if deg <= 0 then return dir end
	local r = math.rad(deg) * math.sqrt(math.random())
	local theta = math.random() * math.pi * 2
	return (CFrame.lookAt(Vector3.zero, dir) * CFrame.Angles(0, 0, theta) * CFrame.Angles(r, 0, 0)).LookVector
end

local function rotateCamera(dPitch, dYaw)
	local cf = camera.CFrame
	if (cf.LookVector.Y > 0.97 and dPitch > 0) or (cf.LookVector.Y < -0.97 and dPitch < 0) then dPitch = 0 end
	local pos = cf.Position
	camera.CFrame = CFrame.Angles(0, math.rad(dYaw), 0) * (cf - pos) * CFrame.Angles(math.rad(dPitch), 0, 0) + pos
end

local function ejectShell()
	local s = Instance.new("Part")
	s.Size = Vector3.new(0.035, 0.035, 0.11)
	s.Color = Color3.fromRGB(214, 168, 72)
	s.Material = Enum.Material.Metal
	s.CanCollide, s.CanQuery, s.CanTouch, s.CastShadow = false, false, false, false
	s.CFrame = gunCF * built.EjectOffset * CFrame.Angles(0, math.rad(90), 0)
	s.Parent = workspace
	local c = camera.CFrame
	s.AssemblyLinearVelocity = root.AssemblyLinearVelocity + c.RightVector * (7 + math.random() * 3)
		+ c.UpVector * (5 + math.random() * 3) - c.LookVector
	s.AssemblyAngularVelocity = Vector3.new(rnd(), rnd(), rnd()) * 30
	Debris:AddItem(s, 0.6)
end

-- ================= RELOAD =================
local AT_MAG = Vector3.new(0.08, -0.33, 0.75)
local MAG_LOW = Vector3.new(0.08, -0.58, 0.8)
local DOWN = Vector3.new(0.35, -2.2, 0.9)
local IDENT = CFrame.identity

local function reloadPose(u)
	if u <= 0 or u >= 1 then return IDENT, IDENT, IDENT end
	local tilt = smooth(seg(u, 0, 0.14)) * (1 - smooth(seg(u, 0.84, 1)))
	local tiltCF = CFrame.new(-0.12 * tilt, 0.08 * tilt, 0.05 * tilt)
		* CFrame.Angles(math.rad(14 * tilt), math.rad(-12 * tilt), math.rad(32 * tilt))
	local hand, grab
	if u < 0.24 then
		local a = smooth(seg(u, 0.06, 0.22)); hand = Vector3.zero:Lerp(AT_MAG, a); grab = a
	elseif u < 0.5 then
		hand = AT_MAG:Lerp(DOWN, smooth(seg(u, 0.26, 0.44))); grab = 1
	elseif u < 0.66 then
		hand = DOWN:Lerp(MAG_LOW, smooth(seg(u, 0.5, 0.64))); grab = 1
	elseif u < 0.76 then
		hand = MAG_LOW:Lerp(AT_MAG, smooth(seg(u, 0.66, 0.7))); grab = 1
	else
		local a = smooth(seg(u, 0.76, 0.9)); hand = AT_MAG:Lerp(Vector3.zero, a); grab = 1 - a
	end
	local leftCF = CFrame.new(hand) * CFrame.Angles(math.rad(-20 * grab), 0, math.rad(-35 * grab))
	local magCF = IDENT
	if u >= 0.24 and u < 0.72 then
		local d = hand - AT_MAG
		local out = math.clamp(d.Magnitude / 0.6, 0, 1)
		magCF = CFrame.new(d) * CFrame.Angles(math.rad(-20 * out), 0, math.rad(12 * out))
	end
	return tiltCF, magCF, leftCF
end

local function startReload()
	if reloading or ammo == cfg.MagSize or humanoid.Health <= 0 then return end
	reloading = true
	reloadStart = os.clock()
	reloadEvents = {
		{ at = 0.02, fn = function() Sfx.Play("ReloadStart") end },
		{ at = 0.26, fn = function() Sfx.Play("MagOut") end },
		{ at = 0.68, fn = function()
			Sfx.Play("MagIn")
			recoilPos:Impulse(Vector3.new(0, 1.5, 0))
			recoilRot:Impulse(Vector3.new(-40, 0, 30))
		end },
		{ at = 0.8, fn = function()
			Sfx.Play("Bolt")
			recoilPos:Impulse(Vector3.new(0, 0, 2.5))
		end },
	}
	ReloadRemote:FireServer(WEAPON)
	updateHUD()
end

-- ================= FIRE =================
local function fire()
	ammo -= 1
	local now = os.clock()
	if now - lastShotTime > cfg.SprayResetTime then sprayIndex = 0 end
	sprayIndex += 1
	lastShotTime = now

	local origin = camera.CFrame.Position
	local dir = spreadDirection(camera.CFrame.LookVector, currentSpread())
	local result = workspace:Raycast(origin, dir * cfg.Range, rayParams)
	local hitPos = result and result.Position or origin + dir * cfg.Range
	local hitPart = result and result.Instance

	fx:Tracer((gunCF * built.MuzzleOffset).Position, hitPos)
	flashUntil = now + 0.035
	flashRoll = math.random() * math.pi * 2
	local fs = 0.8 + math.random() * 0.5
	for _, e in built.Groups.flash do e.Part.Size = e.BaseSize * fs end
	if sprayIndex % 2 == 0 then built.Smoke:Emit(1) end
	ejectShell()
	Sfx.Play("Fire")
	Sfx.Play("FireTail")
	if ammo <= cfg.MagSize * 0.25 then Sfx.Play("LowAmmo") end

	if hitPart then
		local model = hitPart:FindFirstAncestorOfClass("Model")
		if enemyHumanoid(model) then
			local head = GunMath.IsHeadshot(hitPart)
			local dmg = GunMath.Damage(cfg, (hitPos - origin).Magnitude, head)
			fx:Hit(model, hitPos, -dir, math.floor(dmg + 0.5), head)
		elseif not (model and model:FindFirstChildOfClass("Humanoid")) then
			fx:Impact(hitPos, result.Normal, hitPart)
		end
	end

	FireRemote:FireServer(WEAPON, origin, dir, hitPart, hitPos)

	local pat = cfg.RecoilPattern[math.min(sprayIndex, #cfg.RecoilPattern)]
	local m = 1 + (cfg.ADSRecoilMult - 1) * aimAlpha
	climbTarget += Vector2.new(pat.X, pat.Y + rnd() * cfg.RecoilJitter) * m
	camPunch:Impulse(Vector3.new(cfg.CamPunch, rnd() * cfg.CamPunch * 0.35, rnd() * cfg.CamRoll) * m)
	fovPunch:Impulse(Vector3.new(cfg.FovPunch, 0, 0))
	recoilPos:Impulse(Vector3.new(rnd() * 0.4, 0.3, cfg.KickBack))
	recoilRot:Impulse(Vector3.new(cfg.KickUp * (0.85 + math.random() * 0.3), rnd() * cfg.KickYaw, rnd() * cfg.KickRoll))
	bloom = math.min(bloom + cfg.BloomPerShot, cfg.BloomMax)

	updateHUD()
	if ammo <= 0 then startReload() end
end

local function onRender(dt)
	dt = math.min(dt, 0.1)
	local now = os.clock()
	local alive = humanoid.Health > 0
	local grounded = humanoid.FloorMaterial ~= Enum.Material.Air
	local vel = root.AssemblyLinearVelocity
	local flatSpeed = Vector3.new(vel.X, 0, vel.Z).Magnitude
	local moving = humanoid.MoveDirection.Magnitude > 0.1
	local sliding = character:GetAttribute("Sliding") == true
	local mantling = character:GetAttribute("Mantling") == true

	-- states
	local aiming = aimHeld and not reloading and alive and not mantling
	local wantsSprint = shiftHeld and moving and not aiming and not triggerDown
	local sprinting = wantsSprint and not sliding and not mantling
	humanoid.WalkSpeed = wantsSprint and SPRINT_SPEED or WALK_SPEED
	aimAlpha = approach(aimAlpha, aiming and 1 or 0, dt, cfg.ADSTime)
	sprintAlpha = approach(sprintAlpha, sprinting and 1 or 0, dt, 0.18)
	slideAlpha = approach(slideAlpha, sliding and 1 or 0, dt, 0.12)
	mantleAlpha = approach(mantleAlpha, mantling and 1 or 0, dt, 0.1)
	UIS.MouseDeltaSensitivity = 1 + (cfg.ADSSensitivity - 1) * aimAlpha

	-- firing
	local canFire = alive and not reloading and ammo > 0 and sprintAlpha < 0.3 and mantleAlpha < 0.4 and (now - equipStart) > 0.3
	if triggerDown and canFire then
		if nextFireTime < now - INTERVAL then nextFireTime = now end
		while nextFireTime <= now and ammo > 0 do
			fire()
			nextFireTime += INTERVAL
		end
	elseif triggerDown and ammo <= 0 and not reloading then
		startReload()
	end
	bloom = math.max(0, bloom - cfg.BloomRecovery * dt * (triggerDown and 0.25 or 1))

	-- reload progress
	local reloadU = 0
	if reloading then
		local t = now - reloadStart
		reloadU = t / cfg.ReloadTime
		for _, ev in reloadEvents do
			if not ev.done and reloadU >= ev.at then ev.done = true; ev.fn() end
		end
		barFill.BackgroundColor3 = ACCENT
		barFill.Size = UDim2.fromScale(math.clamp(reloadU, 0, 1), 1)
		if t >= cfg.ReloadTime then
			reloading = false
			ammo = cfg.MagSize
			updateHUD()
		end
	end

	-- camera: real climb + visual punch
	local pitch = math.deg(math.asin(math.clamp(camera.CFrame.LookVector.Y, -1, 1)))
	if lastPitch then
		local userDelta = pitch - lastPitch
		if userDelta < 0 and climbTarget.X > 0 then
			local c = math.min(-userDelta, climbTarget.X)
			climbTarget -= Vector2.new(c, 0)
			climbOffset -= Vector2.new(c, 0)
			applied -= Vector2.new(c, 0)
		end
	end
	if now - lastShotTime > cfg.RecoverDelay then
		climbTarget = climbTarget:Lerp(Vector2.zero, 1 - math.exp(-cfg.RecoverSpeed * dt))
	end
	climbOffset = climbOffset:Lerp(climbTarget, 1 - math.exp(-cfg.ClimbSnap * dt))
	local cp = camPunch:Update(dt)
	local desired = climbOffset + Vector2.new(cp.X, cp.Y)
	local d = desired - applied
	applied = desired
	rotateCamera(d.X, d.Y)
	lastPitch = math.deg(math.asin(math.clamp(camera.CFrame.LookVector.Y, -1, 1)))
	local slideRoll = -3 * slideAlpha * (1 - aimAlpha * 0.6)
	camera.CFrame *= CFrame.Angles(0, 0, math.rad(cp.Z + slideRoll))

	local fp = fovPunch:Update(dt)
	local fov = BASE_FOV + (SPRINT_FOV - BASE_FOV) * sprintAlpha + SLIDE_FOV_ADD * slideAlpha
	camera.FieldOfView = fov + (cfg.ADSFov - fov) * aimAlpha + fp.X

	-- ===== viewmodel =====
	local camCF = camera.CFrame
	local md = UIS:GetMouseDelta()
	local swayMult = 1 - aimAlpha * 0.7
	swaySpring.Target = Vector3.new(
		math.clamp(md.Y * 0.35, -5, 5), math.clamp(md.X * 0.35, -5, 5), math.clamp(md.X * 0.25, -4, 4)
	) * swayMult
	local sw = swaySpring:Update(dt)

	local localVel = camCF:VectorToObjectSpace(vel)
	strafeRoll = approach(strafeRoll, -localVel.X * 0.22 * swayMult * (1 - slideAlpha), dt, 0.2)

	if not grounded then airVelY = vel.Y end
	if grounded and not wasGrounded then
		landSpring:Impulse(Vector3.new(0, -math.clamp(-airVelY * 0.1, 0.6, 4), 0))
	elseif not grounded and wasGrounded then
		landSpring:Impulse(Vector3.new(0, -1, 0))
	end
	wasGrounded = grounded
	local lp = landSpring:Update(dt)

	local bobTarget = (grounded and math.clamp(flatSpeed / 16, 0, 1.6) or 0) * (1 - aimAlpha * 0.85) * (1 - slideAlpha) * (1 - mantleAlpha)
	bobAmt = approach(bobAmt, bobTarget, dt, 0.15)
	bobT += dt * flatSpeed * 0.62
	local b = bobAmt
	local bob = CFrame.new(math.sin(bobT) * 0.035 * b, -math.abs(math.cos(bobT)) * 0.04 * b + 0.02 * b, 0)
		* CFrame.Angles(math.rad(math.abs(math.cos(bobT)) * 0.8 * b), 0, math.rad(math.sin(bobT) * 1.4 * b))
	local br = 1 - aimAlpha * 0.8
	local breathe = CFrame.new(0, math.sin(now * 1.7) * 0.01 * br, 0) * CFrame.Angles(math.rad(math.sin(now * 1.7) * 0.3 * br), 0, 0)

	local e = math.clamp((now - equipStart) / 0.45, 0, 1)
	local ie = (1 - e) ^ 3
	local equipCF = CFrame.new(0.1 * ie, -0.9 * ie, 0.2 * ie) * CFrame.Angles(math.rad(-35 * ie), math.rad(-15 * ie), math.rad(20 * ie))

	-- slide: cant left and tuck in (mostly straightened when aiming)
	local sk = smooth(slideAlpha)
	local skPos = sk * (1 - aimAlpha * 0.9)
	local skRoll = sk * (1 - aimAlpha * 0.75)
	local slideCF = CFrame.new(-0.1 * skPos, -0.05 * skPos, 0.12 * skPos)
		* CFrame.Angles(math.rad(3 * skPos), math.rad(4 * skPos), math.rad(-16 * skRoll))

	-- mantle: drop the gun down and to the side
	local mk = smooth(mantleAlpha)
	local mantleCF = CFrame.new(0.15 * mk, -0.75 * mk, 0.3 * mk)
		* CFrame.Angles(math.rad(-38 * mk), math.rad(22 * mk), math.rad(32 * mk))

	local tiltCF, magAnim, leftAnim = reloadPose(reloading and reloadU or 0)

	local rp, rr = recoilPos:Update(dt), recoilRot:Update(dt)
	local kp, kr = 1 - aimAlpha * 0.45, 1 - aimAlpha * 0.6
	local recoilCF = CFrame.new(rp.X * kp, rp.Y * kp, rp.Z * kp) * CFrame.Angles(math.rad(rr.X * kr), math.rad(rr.Y * kr), math.rad(rr.Z * kr))

	local ea = smooth(aimAlpha)
	local base = HIP:Lerp(built.AimOffset, ea):Lerp(SPRINT, sprintAlpha * (1 - aimAlpha))

	gunCF = camCF * base * mantleCF * slideCF * equipCF * CFrame.new(0, lp.Y, 0) * CFrame.Angles(math.rad(-lp.Y * 20), 0, 0)
		* breathe * bob * CFrame.Angles(math.rad(sw.X), math.rad(sw.Y), math.rad(sw.Z + strafeRoll))
		* tiltCF * recoilCF

	local i = 0
	for _, g in built.Groups.gun do i += 1; cfList[i] = gunCF * g.Offset end
	local magCF = gunCF * built.MagPivot * magAnim
	for _, g in built.Groups.mag do i += 1; cfList[i] = magCF * g.Offset end
	local leftCF = gunCF * built.LeftPivot * leftAnim
	for _, g in built.Groups.left do i += 1; cfList[i] = leftCF * g.Offset end
	local flashCF = gunCF * built.MuzzleOffset * CFrame.Angles(0, 0, flashRoll)
	for _, g in built.Groups.flash do i += 1; cfList[i] = flashCF * g.Offset end
	workspace:BulkMoveTo(partsList, cfList, Enum.BulkMoveMode.FireCFrameChanged)

	local flashOn = now < flashUntil
	if flashOn ~= flashVisible then
		flashVisible = flashOn
		for _, g in built.Groups.flash do g.Part.Transparency = flashOn and 0 or 1 end
		built.Light.Brightness = flashOn and 4 or 0
	end

	-- crosshair
	local spread = currentSpread()
	local gap = math.tan(math.rad(spread)) / math.tan(math.rad(camera.FieldOfView / 2)) * (camera.ViewportSize.Y / 2) + 5
	gap = math.min(gap, 110)
	for _, c in crossLines do
		c.f.Position = UDim2.new(0.5, c.d.X * (gap + 4), 0.5, c.d.Y * (gap + 4))
	end
	cross.GroupTransparency = math.clamp(math.max(aimAlpha, sprintAlpha, mantleAlpha) * 1.3, 0, 1)

	-- damage vignette
	dmgAlpha = math.max(0, dmgAlpha - dt * 2.2)
	local low = humanoid.Health < 50 and (1 - humanoid.Health / 50) * 0.5 or 0
	local va = math.max(dmgAlpha, low)
	for _, f in vignettes do f.BackgroundTransparency = 1 - va * 0.6 end
end

local function rootHalf() return root.Size.Y / 2 end
local function feetY() return root.Position.Y - hum.HipHeight - rootHalf() end
local function grounded() return hum.FloorMaterial ~= Enum.Material.Air end
local function hvel() return flat(root.AssemblyLinearVelocity) end
local function flatLook()
	local l = flat(root.CFrame.LookVector)
	return l.Magnitude > 0.01 and l.Unit or Vector3.new(0, 0, -1)
end
local function charHeight() return baseHip + root.Size.Y + 1.3 end

local function findLedge()
	refreshFilters()
	local dir = flatLook()
	local fy = feetY()
	local wall
	for _, h in ipairs({ 0.8, 1.8, 2.8, 3.8 }) do
		local r = workspace:Raycast(Vector3.new(root.Position.X, fy + h, root.Position.Z), dir * CFG.MANTLE_REACH, rayParams)
		if r and math.abs(r.Normal.Y) < 0.5 then wall = r break end
	end
	if not wall then return nil end
	local inward = flat(-wall.Normal)
	if inward.Magnitude < 0.1 then return nil end
	inward = inward.Unit

	local probe = wall.Position + inward * 0.7
	local start = Vector3.new(probe.X, fy + CFG.MANTLE_MAX_H + 0.6, probe.Z)
	if #workspace:GetPartBoundsInRadius(start, 0.2, overlapParams) > 0 then return nil end -- wall too tall
	local down = workspace:Raycast(start, Vector3.new(0, -(CFG.MANTLE_MAX_H + 0.6), 0), rayParams)
	if not down or down.Normal.Y < 0.7 then return nil end
	local h = down.Position.Y - fy
	if h < CFG.MANTLE_MIN_H or h > CFG.MANTLE_MAX_H then return nil end

	local standXZ = down.Position + inward * 0.5
	local height = charHeight()
	local box = CFrame.new(standXZ.X, down.Position.Y + 0.15 + height / 2, standXZ.Z)
	if #workspace:GetPartBoundsInBox(box, Vector3.new(1.8, height, 1.4), overlapParams) > 0 then return nil end

	local lipY = down.Position.Y + 0.6
	local from = Vector3.new(root.Position.X, lipY, root.Position.Z)
	if workspace:Raycast(from, Vector3.new(standXZ.X, lipY, standXZ.Z) - from, rayParams) then return nil end
	local upLen = math.max(down.Position.Y + height - root.Position.Y, 0.1)
	if workspace:Raycast(root.Position, Vector3.new(0, upLen, 0), rayParams) then return nil end

	return {
		top = down.Position.Y, inward = inward, height = h,
		stand = Vector3.new(standXZ.X, down.Position.Y + baseHip + rootHalf() + 0.05, standXZ.Z),
	}
end

------------------------------------------------------------------ state
local mode = "none" -- none | slide | carry | mantle
local slideDir, slideSpeed, slideTime, slideAir, slideStuck = Vector3.zero, 0, 0, 0, 0
local slideReadyAt, slideQueuedAt = 0, -1
local carryV, carryT = Vector3.zero, 0
local mantleData, mantleReadyAt = nil, 0
local lastJumpReq, lastLedgeCheck = -1, 0
local wasGrounded = true
local camKick, camOffsetY = 0, 0

local function canSlide()
	return mode == "none" and hum.Health > 0 and grounded() and os.clock() >= slideReadyAt
		and hvel().Magnitude >= CFG.SLIDE_MIN_SPEED
end

local function startSlide()
	refreshFilters()
	local v = hvel()
	slideDir = v.Magnitude > 0.1 and v.Unit or flatLook()
	slideSpeed = math.min(v.Magnitude + CFG.SLIDE_BOOST, CFG.SLIDE_MAX_START)
	slideTime, slideAir, slideStuck = 0, 0, 0
	mode = "slide"
	char:SetAttribute("Sliding", true)
	playSound("Slide", 0.7)
	drivePlanar(slideDir * slideSpeed)
end

local function endSlide()
	mode = "none"
	release()
	char:SetAttribute("Sliding", false)
	slideReadyAt = os.clock() + CFG.SLIDE_COOLDOWN
end

local function finishMantle()
	release()
	if mantleData then root.AssemblyLinearVelocity = mantleData.ledge.inward * CFG.MANTLE_EXIT_SPEED end
	mantleData = nil
	mode = "none"
	char:SetAttribute("Mantling", false)
	mantleReadyAt = os.clock() + CFG.MANTLE_COOLDOWN
end

local function startMantle(ledge)
	if mode == "slide" then endSlide() end
	release()
	local riseY = ledge.top + baseHip + rootHalf() + 0.25
	mantleData = {
		ledge = ledge, phase = 1, t = 0, total = 0, riseY = riseY,
		upTime = clamp(0.1 + math.max(riseY - root.Position.Y, 0) * 0.045, 0.12, 0.4),
	}
	mode = "mantle"
	char:SetAttribute("Mantling", true)
	camKick = -0.35
	playSound("Mantle", 0.6)
end

------------------------------------------------------------------ updates
local function updateSlide(dt)
	slideTime += dt
	local r = workspace:Raycast(root.Position, Vector3.new(0, -(hum.HipHeight + rootHalf() + 1.5), 0), rayParams)
	if r then
		slideAir = 0
		slideSpeed += flat(r.Normal):Dot(slideDir) * CFG.SLIDE_SLOPE_ACCEL * dt
	else
		slideAir += dt
	end
	slideSpeed -= CFG.SLIDE_FRICTION * dt

	local md = flat(hum.MoveDirection)
	if md.Magnitude > 0.1 then
		local target = md.Unit
		local angle = math.acos(clamp(slideDir:Dot(target), -1, 1))
		if angle > 1e-3 and angle < math.rad(120) then
			slideDir = slideDir:Lerp(target, math.min(1, CFG.SLIDE_STEER * dt / angle)).Unit
		end
	end
	drivePlanar(slideDir * math.max(slideSpeed, 0))

	if slideTime > 0.1 and hvel().Magnitude < slideSpeed * 0.4 then slideStuck += dt else slideStuck = 0 end
	if slideSpeed <= CFG.SLIDE_END_SPEED or slideTime >= CFG.SLIDE_MAX_TIME or slideAir > 0.25 or slideStuck > 0.1 then
		endSlide()
	end
end

local function updateCarry(dt)
	carryT += dt
	local f = 1 - clamp(carryT / CFG.SLIDE_JUMP_CARRY, 0, 1)
	local steer = flat(hum.MoveDirection) * hum.WalkSpeed
	drivePlanar(carryV * f + steer * (1 - f))
	if carryT >= CFG.SLIDE_JUMP_CARRY or (grounded() and carryT > 0.15) then
		release()
		mode = "none"
	end
end

local function updateMantle(dt)
	local m = mantleData
	m.t += dt
	m.total += dt
	if m.phase == 1 then
		local remain = math.max(m.upTime - m.t, dt)
		local vy = math.max((m.riseY - root.Position.Y) / remain, 4)
		driveVector(Vector3.new(0, vy, 0) + m.ledge.inward * 1.5)
		if root.Position.Y >= m.riseY - 0.08 or m.t > m.upTime + 0.15 then
			m.phase, m.t = 2, 0
		end
	else
		local remain = math.max(CFG.MANTLE_FWD_TIME - m.t, dt)
		driveVector((m.ledge.stand - root.Position) / remain)
		if m.t >= CFG.MANTLE_FWD_TIME then finishMantle() return end
	end
	if m.total > 1 then finishMantle() end
end

------------------------------------------------------------------ input
local function onSlide(_, state)
	if state ~= Enum.UserInputState.Begin then return Enum.ContextActionResult.Pass end
	if canSlide() then startSlide() else slideQueuedAt = os.clock() end
	return Enum.ContextActionResult.Sink
end
CAS:BindAction("Slide", onSlide, true, Enum.KeyCode.C, Enum.KeyCode.LeftControl, Enum.KeyCode.ButtonB)
CAS:SetTitle("Slide", "Slide")

local jumpConn = UIS.JumpRequest:Connect(function()
	lastJumpReq = os.clock()
	if hum.Health <= 0 then return end
	if mode == "slide" or (mode == "none" and grounded()) then
		if os.clock() >= mantleReadyAt then
			local ledge = findLedge()
			if ledge and ledge.height >= CFG.MANTLE_GROUND_MIN then startMantle(ledge) return end
		end
		if mode == "slide" then -- slide-jump keeps momentum
			local keep = slideDir * math.max(slideSpeed, 0) * CFG.SLIDE_JUMP_KEEP
			endSlide()
			carryV, carryT = keep, 0
			mode = "carry"
			hum:ChangeState(Enum.HumanoidStateType.Jumping)
			drivePlanar(carryV)
		end
	end
end)

------------------------------------------------------------------ main loop
local stepConn = RunService.Stepped:Connect(function(_, dt)
	if hum.Health <= 0 then return end
	local now = os.clock()
	local g = grounded()
	if g and not wasGrounded and now - slideQueuedAt < CFG.SLIDE_BUFFER and canSlide() then startSlide() end
	wasGrounded = g

	if mode == "slide" then updateSlide(dt)
	elseif mode == "carry" then updateCarry(dt)
	elseif mode == "mantle" then updateMantle(dt) end

	-- mantle while airborne: holding jump and moving toward a ledge
	if (mode == "none" or mode == "carry") and not g and now >= mantleReadyAt and now - lastLedgeCheck > 0.05 then
		local jumpHeld = UIS:IsKeyDown(Enum.KeyCode.Space)
			or UIS:IsGamepadButtonDown(Enum.UserInputType.Gamepad1, Enum.KeyCode.ButtonA)
			or now - lastJumpReq < CFG.MANTLE_JUMP_HOLD
		local md = flat(hum.MoveDirection)
		if jumpHeld and md.Magnitude > 0.1 and md.Unit:Dot(flatLook()) > 0.3 then
			lastLedgeCheck = now
			local ledge = findLedge()
			if ledge then startMantle(ledge) end
		end
	end

	-- hip height (slide crouch), stays low under ceilings
	local targetHip = (mode == "slide") and baseHip * CFG.SLIDE_HIP_FACTOR or baseHip
	if targetHip > hum.HipHeight + 0.01 then
		local need = (targetHip - hum.HipHeight) + rootHalf() + 1.4
		if workspace:Raycast(root.Position, Vector3.new(0, need, 0), rayParams) then targetHip = hum.HipHeight end
	end
	if math.abs(targetHip - hum.HipHeight) > 0.005 then setHip(approach(hum.HipHeight, targetHip, 20, dt)) end

	-- camera
	camKick = approach(camKick, 0, 10, dt)
	camOffsetY = approach(camOffsetY, (mode == "slide" and CFG.CAM_SLIDE_DROP or 0) + camKick, 14, dt)
	hum.CameraOffset = Vector3.new(0, camOffsetY, 0)
end)

hum.Died:Connect(function()
	release()
	char:SetAttribute("Sliding", false)
	char:SetAttribute("Mantling", false)
	CAS:UnbindAction("Slide")
	jumpConn:Disconnect()
	stepConn:Disconnect()
end)

local TweenService = game:GetService("TweenService")
local RunService = game:GetService("RunService")
local Debris = game:GetService("Debris")
local Sfx = require(script.Parent:WaitForChild("Sfx"))

local HitFX = {}
HitFX.__index = HitFX

local WHITE = Color3.fromRGB(255, 255, 255)
local HEAD = Color3.fromRGB(255, 212, 60)
local KILL = Color3.fromRGB(255, 70, 70)
local SPARK_TEX = "rbxasset://textures/particles/sparkles_main.dds"
local SMOKE_TEX = "rbxasset://textures/particles/smoke_main.dds"

local function escape(s)
	return (s:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;"))
end

local function anchorPos(model)
	local p = model:FindFirstChild("Head") or model.PrimaryPart or model:FindFirstChild("HumanoidRootPart")
	return (p and p:IsA("BasePart")) and p.Position or nil
end

local function makeLabel(parent, text, color, size)
	local l = Instance.new("TextLabel")
	l.AnchorPoint = Vector2.new(0.5, 0.5)
	l.Size = UDim2.fromOffset(140, 50)
	l.BackgroundTransparency = 1
	l.Font = Enum.Font.GothamBlack
	l.TextSize = size
	l.TextColor3 = color
	l.Text = text
	local st = Instance.new("UIStroke")
	st.Thickness = 2; st.Color = Color3.new(0, 0, 0); st.Transparency = 0.25
	st.Parent = l
	local sc = Instance.new("UIScale")
	sc.Parent = l
	l.Parent = parent
	return l, sc, st
end

function HitFX.new(gui)
	local self = setmetatable({}, HitFX)
	self.gui = gui
	self.camera = workspace.CurrentCamera
	self.popups, self.combos, self.highlights, self.holes = {}, {}, {}, {}

	self.folder = Instance.new("Folder")
	self.folder.Name = "ClientFX"
	self.folder.Parent = workspace

	local anchor = Instance.new("Part")
	anchor.Anchored = true
	anchor.CanCollide, anchor.CanQuery, anchor.CanTouch = false, false, false
	anchor.Transparency = 1
	anchor.Size = Vector3.one * 0.1
	anchor.Parent = self.folder
	self.anchor = anchor

	self.layer = Instance.new("Frame")
	self.layer.Size = UDim2.fromScale(1, 1)
	self.layer.BackgroundTransparency = 1
	self.layer.Parent = gui

	self:_buildHitmarker()
	self:_buildBanner()
	self.conn = RunService.RenderStepped:Connect(function(dt) self:_update(dt) end)
	return self
end

function HitFX:Destroy()
	self.conn:Disconnect()
	self.folder:Destroy()
	for _, h in self.highlights do h:Destroy() end
end

-- ===== hitmarker =====
function HitFX:_buildHitmarker()
	local hm = Instance.new("Frame")
	hm.AnchorPoint = Vector2.new(0.5, 0.5)
	hm.Position = UDim2.fromScale(0.5, 0.5)
	hm.Size = UDim2.fromOffset(40, 40)
	hm.BackgroundTransparency = 1
	hm.Visible = false
	hm.Parent = self.gui
	local scale = Instance.new("UIScale")
	scale.Parent = hm
	self.hmLines = {}
	for _, o in { Vector2.new(-1, -1), Vector2.new(1, -1), Vector2.new(-1, 1), Vector2.new(1, 1) } do
		local l = Instance.new("Frame")
		l.AnchorPoint = Vector2.new(0.5, 0.5)
		l.BorderSizePixel = 0
		l.Rotation = (o.X * o.Y > 0) and -45 or 45
		local st = Instance.new("UIStroke")
		st.Transparency = 0.4
		st.Parent = l
		l.Parent = hm
		table.insert(self.hmLines, { f = l, o = o })
	end
	self.hm, self.hmScale, self.hmToken = hm, scale, 0
end

function HitFX:ShowHitmarker(kind)
	local color = kind == "kill" and KILL or kind == "head" and HEAD or WHITE
	local w, h, dist = 3, 10, 9
	if kind == "head" then h, dist = 12, 10 elseif kind == "kill" then w, h, dist = 4, 16, 13 end
	for _, l in self.hmLines do
		l.f.BackgroundColor3 = color
		l.f.Size = UDim2.fromOffset(w, h)
		l.f.Position = UDim2.new(0.5, l.o.X * dist, 0.5, l.o.Y * dist)
	end
	self.hmToken += 1
	local token = self.hmToken
	self.hm.Visible = true
	self.hmScale.Scale = kind == "kill" and 2.0 or 1.6
	TweenService:Create(self.hmScale, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Scale = 1 }):Play()
	local dur = kind == "kill" and 0.45 or kind == "head" and 0.18 or 0.12
	task.delay(dur, function()
		if self.hmToken == token then self.hm.Visible = false end
	end)
end

-- ===== elimination banner =====
function HitFX:_buildBanner()
	local b = Instance.new("TextLabel")
	b.AnchorPoint = Vector2.new(0.5, 0.5)
	b.Position = UDim2.fromScale(0.5, 0.64)
	b.Size = UDim2.fromOffset(600, 40)
	b.BackgroundTransparency = 1
	b.Font = Enum.Font.GothamBlack
	b.TextSize = 26
	b.TextColor3 = KILL
	b.RichText = true
	b.TextTransparency = 1
	b.TextStrokeTransparency = 1
	local s = Instance.new("UIScale")
	s.Parent = b
	b.Parent = self.gui
	self.banner, self.bannerScale, self.bannerToken = b, s, 0
end

function HitFX:_showBanner(name)
	self.bannerToken += 1
	local token = self.bannerToken
	self.banner.Text = string.format('ELIMINATED  <font color="#FFFFFF">%s</font>', escape(string.upper(name)))
	self.banner.TextTransparency = 0
	self.banner.TextStrokeTransparency = 0.5
	self.bannerScale.Scale = 1.5
	TweenService:Create(self.bannerScale, TweenInfo.new(0.22, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Scale = 1 }):Play()
	task.delay(1.4, function()
		if self.bannerToken ~= token then return end
		TweenService:Create(self.banner, TweenInfo.new(0.3), { TextTransparency = 1, TextStrokeTransparency = 1 }):Play()
	end)
end

-- ===== particles =====
function HitFX:_burst(pos, dir, count, p)
	if dir.Magnitude < 1e-3 then dir = Vector3.yAxis end
	local att = Instance.new("Attachment")
	att.Parent = self.anchor
	att.WorldCFrame = CFrame.lookAt(pos, pos + dir.Unit)
	local e = Instance.new("ParticleEmitter")
	e.Enabled = false
	e.Texture = p.Texture or SPARK_TEX
	e.Color = ColorSequence.new(p.Color or WHITE)
	e.LightEmission = p.LightEmission or 1
	e.LightInfluence = p.LightInfluence or 0
	e.Size = p.Size or NumberSequence.new(0.2, 0)
	e.Transparency = p.Transparency or NumberSequence.new(0, 1)
	e.Lifetime = p.Lifetime or NumberRange.new(0.15, 0.3)
	e.Speed = p.Speed or NumberRange.new(15, 30)
	e.SpreadAngle = p.Spread or Vector2.new(45, 45)
	e.Drag = p.Drag or 0
	e.Acceleration = p.Accel or Vector3.zero
	e.Rotation = NumberRange.new(0, 360)
	e.RotSpeed = p.RotSpeed or NumberRange.new(0, 0)
	e.EmissionDirection = Enum.NormalId.Front
	if p.Streak then e.Orientation = Enum.ParticleOrientation.VelocityParallel end
	e.Parent = att
	e:Emit(count)
	Debris:AddItem(att, (p.Lifetime and p.Lifetime.Max or 0.3) + 1)
	return att
end

function HitFX:_flash(model, color)
	local h = self.highlights[model]
	if not h or not h.Parent then
		h = Instance.new("Highlight")
		h.DepthMode = Enum.HighlightDepthMode.Occluded
		h.OutlineTransparency = 1
		h.FillTransparency = 1
		h.Adornee = model
		h.Parent = model
		self.highlights[model] = h
	end
	h.FillColor = color
	h.FillTransparency = 0.2
	TweenService:Create(h, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { FillTransparency = 1 }):Play()
end

function HitFX:Tracer(from, to)
	local dist = (to - from).Magnitude
	if dist < 1 then return end
	local dir = (to - from).Unit
	local len = math.min(dist, 7)
	local p = Instance.new("Part")
	p.Anchored = true
	p.CanCollide, p.CanQuery, p.CanTouch, p.CastShadow = false, false, false, false
	p.Material = Enum.Material.Neon
	p.Color = Color3.fromRGB(255, 214, 150)
	p.Size = Vector3.new(0.045, 0.045, len)
	p.CFrame = CFrame.lookAt(from + dir * len * 0.5, to)
	p.Parent = self.folder
	local tw = TweenService:Create(p, TweenInfo.new(math.clamp(dist / 1100, 0.02, 0.4), Enum.EasingStyle.Linear),
		{ CFrame = CFrame.lookAt(to - dir * len * 0.5, to + dir) })
	tw:Play()
	tw.Completed:Once(function() p:Destroy() end)
end

-- ===== damage numbers =====
function HitFX:_popup(worldPos, dmg, head)
	local l, sc, st = makeLabel(self.layer, tostring(dmg), head and HEAD or WHITE, head and 30 or 24)
	l.Rotation = math.random(-10, 4)
	table.insert(self.popups, {
		label = l, scale = sc, stroke = st, world = worldPos, age = 0, life = 0.75,
		offset = Vector2.new(34 + math.random() * 16, -10 + math.random() * 14),
		vel = Vector2.new(150 + math.random() * 90, -230 - math.random() * 60),
	})
end

function HitFX:_combo(model, dmg, head)
	local now = os.clock()
	local c = self.combos[model]
	if not c or now - c.last > 1.2 or c.killed then
		if c then c.label:Destroy() end
		local l, sc, st = makeLabel(self.layer, "", WHITE, 34)
		c = { label = l, scale = sc, stroke = st, total = 0, last = now, killed = false }
		self.combos[model] = c
	end
	c.total += dmg
	c.last = now
	c.label.Text = tostring(c.total)
	c.label.TextColor3 = head and HEAD or WHITE
	c.scale.Scale = 1.35
end

	self:_combo(model, dmg, head)
	self:_flash(model, head and HEAD or WHITE)
	self:_burst(pos, normal, head and 12 or 7, {
		Color = head and HEAD or Color3.fromRGB(255, 240, 220), Size = NumberSequence.new(0.14, 0),
		Speed = NumberRange.new(14, 28), Lifetime = NumberRange.new(0.1, 0.22),
		Accel = Vector3.new(0, -50, 0), Streak = true,
	})

		Color = head and HEAD or WHITE, Size = NumberSequence.new(head and 1.6 or 1.1, 0),
		Speed = NumberRange.new(0, 0), Lifetime = NumberRange.new(0.07, 0.07),
	})
	self:ShowHitmarker(head and "head" or "body")
	Sfx.Play(head and "Headshot" or "Hit")
end

-- ===== per-frame =====
function HitFX:_update(dt)
	local cam = self.camera
	local now = os.clock()

	for i = #self.popups, 1, -1 do
		local p = self.popups[i]
		p.age += dt
		local t = p.age
		if t >= p.life then
			p.label:Destroy()
			table.remove(self.popups, i)
			continue
		end
		local sp = cam:WorldToViewportPoint(p.world)
		if sp.Z <= 0 then
			p.label.Visible = false
		else
			p.label.Visible = true
			local off = p.offset + p.vel * t + Vector2.new(0, 520) * (t * t * 0.5)
			p.label.Position = UDim2.fromOffset(sp.X + off.X, sp.Y + off.Y)
		end
		local pop = math.clamp(t / 0.09, 0, 1)
		p.scale.Scale = 1 + 0.9 * (1 - pop) * (1 - pop)
		local fade = math.clamp((t - (p.life - 0.25)) / 0.25, 0, 1)
		p.label.TextTransparency = fade
		p.stroke.Transparency = 0.25 + 0.75 * fade
	end

	for model, c in self.combos do
		local since = now - c.last
		local life = c.killed and 1.6 or 1.2
		if since > life then
			c.label:Destroy()
			self.combos[model] = nil
			continue
		end
		local pos = (model.Parent and anchorPos(model)) or c.lastPos
		c.lastPos = pos
		if pos then
			local sp = cam:WorldToViewportPoint(pos)
			c.label.Visible = sp.Z > 0
			c.label.Position = UDim2.fromOffset(sp.X + 70, sp.Y - 45)
		end
		c.scale.Scale += (1 - c.scale.Scale) * (1 - math.exp(-dt * 18))
		local fade = math.clamp((since - (life - 0.3)) / 0.3, 0, 1)
		c.label.TextTransparency = fade
		c.stroke.Transparency = 0.25 + 0.75 * fade
	end
end

return HitFX

-- // PATH: ReplicatedStorage.Spring // --
local Spring = {}
Spring.__index = Spring

function Spring.new(stiffness, damping)
	return setmetatable({
		Target = Vector3.zero, Position = Vector3.zero, Velocity = Vector3.zero,
		Stiffness = stiffness or 200, Damping = damping or 20,
	}, Spring)
end

function Spring:Impulse(v)
	self.Velocity += v
end

function Spring:Update(dt)
	local steps = math.max(1, math.ceil(dt / (1 / 240)))
	local h = dt / steps
	for _ = 1, steps do
		local force = (self.Target - self.Position) * self.Stiffness - self.Velocity * self.Damping
		self.Velocity += force * h
		self.Position += self.Velocity * h
	end
	return self.Position
end

return Spring


-- // PATH: ReplicatedStorage.GunMath // --
local GunMath = {}

function GunMath.Falloff(cfg, dist)
	if dist <= cfg.FalloffStart then return 1 end
	if dist >= cfg.FalloffEnd then return cfg.FalloffMin end
	local a = (dist - cfg.FalloffStart) / (cfg.FalloffEnd - cfg.FalloffStart)
	return 1 + (cfg.FalloffMin - 1) * a
end

function GunMath.IsHeadshot(part)
	if part.Name == "Head" then return true end
	local acc = part.Parent
	return acc ~= nil and acc:IsA("Accessory") and (acc.AccessoryType == Enum.AccessoryType.Hat
		or acc.AccessoryType == Enum.AccessoryType.Hair or acc.AccessoryType == Enum.AccessoryType.Face)
end

function GunMath.Damage(cfg, dist, head)
	return cfg.Damage * GunMath.Falloff(cfg, dist) * (head and cfg.HeadMultiplier or 1)
end

return GunMath

local Players = game:GetService("Players")
local RS = game:GetService("ReplicatedStorage")

local remotes = Instance.new("Folder")
remotes.Name = "GunRemotes"
local function remote(name)
	local r = Instance.new("RemoteEvent"); r.Name = name; r.Parent = remotes; return r
end
local FireRemote, ReloadRemote = remote("Fire"), remote("Reload")
local HitConfirm, ShotFX, Damaged = remote("HitConfirm"), remote("ShotFX"), remote("Damaged")
remotes.Parent = RS

local state = {}
local function getState(player, weapon)
	state[player] = state[player] or {}
	local w = state[player][weapon]
	if not w then
		w = { ammo = Config[weapon].MagSize, lastFire = 0, reloadUntil = 0 }
		state[player][weapon] = w
	end
	return w
end
Players.PlayerAdded:Connect(function(p) p.CharacterAdded:Connect(function() state[p] = nil end) end)
Players.PlayerRemoving:Connect(function(p) state[p] = nil end)

ReloadRemote.OnServerEvent:Connect(function(player, weapon)
	local cfg = Config[weapon]; if not cfg then return end
	local w = getState(player, weapon)
	local now = os.clock()
	if now < w.reloadUntil then return end
	w.reloadUntil = now + cfg.ReloadTime * 0.9
	w.ammo = cfg.MagSize
end)

FireRemote.OnServerEvent:Connect(function(player, weapon, origin, dir, hitPart, hitPos)
	local cfg = Config[weapon]
	if not cfg or typeof(origin) ~= "Vector3" or typeof(dir) ~= "Vector3" then return end
	local m = dir.Magnitude
	if not (m > 0.9 and m < 1.1) then return end

	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local myHum = char and char:FindFirstChildOfClass("Humanoid")
	if not root or not myHum or myHum.Health <= 0 then return end
	if (origin - root.Position).Magnitude > 10 then return end

	local w = getState(player, weapon)
	local now = os.clock()
	if now < w.reloadUntil or w.ammo <= 0 then return end
	if now - w.lastFire < (60 / cfg.RPM) * 0.5 then return end
	w.ammo -= 1
	w.lastFire = now

	-- show this shot to everyone else
	local endPos = origin + dir.Unit * cfg.Range
	if typeof(hitPos) == "Vector3" and (hitPos - origin).Magnitude <= cfg.Range + 5 then endPos = hitPos end
	for _, p in Players:GetPlayers() do
		if p ~= player then ShotFX:FireClient(p, player, origin, endPos) end
	end

	if typeof(hitPart) ~= "Instance" or not hitPart:IsA("BasePart") or typeof(hitPos) ~= "Vector3" then return end
	if not hitPart:IsDescendantOf(workspace) then return end

	local model = hitPart:FindFirstAncestorOfClass("Model")
	local hum = model and model:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 or model == char then return end

	local targetPlayer = Players:GetPlayerFromCharacter(model)
	if targetPlayer and not player.Neutral and targetPlayer.Team == player.Team then return end

	local toHit = hitPos - origin
	local dist = toHit.Magnitude
	if dist > cfg.Range or dist < 0.01 then return end
	if toHit.Unit:Dot(dir.Unit) < 0.95 then return end
	if (hitPart.Position - hitPos).Magnitude > hitPart.Size.Magnitude / 2 + 7 then return end

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	local ignore = {}
	for _, p in Players:GetPlayers() do if p.Character then table.insert(ignore, p.Character) end end
	table.insert(ignore, model)
	params.FilterDescendantsInstances = ignore
	if workspace:Raycast(origin, toHit.Unit * math.max(dist - 0.5, 0), params) then return end

	local head = GunMath.IsHeadshot(hitPart)
	local dmg = GunMath.Damage(cfg, dist, head)
	hum:TakeDamage(dmg)
	HitConfirm:FireClient(player, model, head, hum.Health <= 0, math.floor(dmg + 0.5))
	if targetPlayer then Damaged:FireClient(targetPlayer, dmg, origin) end
end)

local lastLook = {}
Look.OnServerEvent:Connect(function(player, pitch, ads, sprint, slide, mantle)
	if type(pitch) ~= "number" or pitch ~= pitch then return end
	local now = os.clock()
	if now - (lastLook[player] or 0) < 1 / 40 then return end
	lastLook[player] = now
	local char = alive(player)
	if not char then return end
	pitch = math.clamp(pitch, -math.rad(80), math.rad(80))
	ads, slide, mantle = ads == true, slide == true, mantle == true
	char:SetAttribute("AimPitch", math.floor(pitch * 100 + 0.5) / 100)
	char:SetAttribute("ADS", ads and not mantle)
	char:SetAttribute("Sprint", sprint == true and not ads and not slide)
	char:SetAttribute("Sliding", slide)
	char:SetAttribute("Mantling", mantle)
end)

local function findRemote(name)
	local deadline = os.clock() + 20
	repeat
		for _, d in ipairs(RS:GetDescendants()) do
			if d.Name == name and d:IsA("RemoteEvent") and not d:IsDescendantOf(folder) then return d end
		end
		task.wait(0.25)
	until os.clock() > deadline
	warn("[ThirdPersonServer] Could not find RemoteEvent '" .. name .. "'")
end

local lastShot = {}
task.spawn(function()
	local fire = findRemote("Fire")
	if not fire then return end
	fire.OnServerEvent:Connect(function(player)
		local char = alive(player)
		if not char then return end
		local now = os.clock()
		if now - (lastShot[player] or 0) < 0.045 then return end
		lastShot[player] = now
		Shot:FireAllClients(char)
	end)
end)

local reloadToken = {}
task.spawn(function()
	local reload = findRemote("Reload")
	if not reload then return end
	reload.OnServerEvent:Connect(function(player)
		local char = alive(player)
		if not char or char:GetAttribute("Reloading") then return end
		local token = {}
		reloadToken[player] = token
		char:SetAttribute("Reloading", true)
		task.delay(RELOAD_TIME, function()
			if reloadToken[player] == token and char.Parent then char:SetAttribute("Reloading", false) end
		end)
	end)
end)

Players.PlayerRemoving:Connect(function(p)
	lastLook[p], lastShot[p], reloadToken[p] = nil, nil, nil
end)
